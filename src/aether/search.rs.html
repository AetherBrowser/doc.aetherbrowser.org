5//! Search engines configured for the shell.
6//!
7//! Firefox stores this choice in the profile as `search.json.mozlz4`: the
8//! configured engines, plus `defaultEngineId` (user choice) falling back to
9//! `appDefaultEngineId`. Built-in engines are recorded by id. An engine the
10//! user added also stores its URL template (`{searchTerms}`).
11//!
12//! Aether keeps the same split in plain JSON at `{config_dir}/search.json`.
13//! `{searchTerms}` is accepted and stored as `%s`, which the location bar
14//! already substitutes. Changes from `servo:settings#search` are written back
15//! to that file.
16

61/// One configured search engine, built-in or added by the user.
62#[derive(Clone, Debug, PartialEq, Eq)]
63pub(crate) struct SearchEngine {
64    pub id: String,
65    pub name: String,
66    /// Search URL template. `%s` stands in for the query.
67    pub url: String,
68    pub alias: Option<String>,
69    pub order: Option<u32>,
70    /// `true` when this engine comes from [`builtin_engines`] and has no custom URL.
71    pub builtin: bool,
72}

74/// The engines configured for this profile, and which one is the default.
75#[derive(Clone, Debug, PartialEq, Eq)]
76pub(crate) struct SearchEngines {
77    engines: Vec<SearchEngine>,
78    default_id: String,
79}

81impl SearchEngines {
82    /// Every built-in engine, with DuckDuckGo as the default.
83    ///
84    /// Used when `search.json` is missing or cannot be read.
85    pub(crate) fn builtin() -> Self {
86        let engines = BUILTIN_ENGINES
87            .iter()
88            .map(|engine| SearchEngine {
89                id: engine.id.to_owned(),
90                name: engine.name.to_owned(),
91                url: engine.url.to_owned(),
92                alias: None,
93                order: None,
94                builtin: true,
95            })
96            .collect();
97        Self {
98            engines,
99            default_id: APP_DEFAULT_ENGINE_ID.to_owned(),
100        }
101    }
102
103    /// Engines the application knows how to search with, whether or not the
104    /// user has them in `search.json`.
105    #[cfg(test)]
106    pub(crate) fn builtin_catalog() -> &'static [BuiltinEngine] {
107        BUILTIN_ENGINES
108    }
109
110    /// Every configured engine, in display order.
111    #[cfg(test)]
112    pub(crate) fn engines(&self) -> &[SearchEngine] {
113        &self.engines
114    }
115
116    pub(crate) fn default_engine(&self) -> &SearchEngine {
117        self.engines
118            .iter()
119            .find(|engine| engine.id == self.default_id)
120            .or_else(|| self.engines.first())
121            .expect("search engines are non-empty")
122    }
123
124    /// Configured engines other than the default.
125    #[cfg(test)]
126    pub(crate) fn additional_engines(&self) -> impl Iterator<Item = &SearchEngine> {
127        let default_id = self.default_id.clone();
128        self.engines
129            .iter()
130            .filter(move |engine| engine.id != default_id)
131    }
132
133    pub(crate) fn set_default(&mut self, id: &str) -> Result<(), String> {
134        if !self.engines.iter().any(|engine| engine.id == id) {
135            return Err(format!("Unknown search engine {id}"));
136        }
137        self.default_id = id.to_owned();
138        Ok(())
139    }
140
141    pub(crate) fn remove(&mut self, id: &str) -> Result<(), String> {
142        if self.engines.len() <= 1 {
143            return Err("At least one search engine is required".to_owned());
144        }
145        let Some(index) = self.engines.iter().position(|engine| engine.id == id) else {
146            return Err(format!("Unknown search engine {id}"));
147        };
148        self.engines.remove(index);
149        if self.default_id == id {
150            self.default_id = self.engines[0].id.clone();
151        }
152        Ok(())
153    }
154
155    pub(crate) fn add_builtin(&mut self, id: &str) -> Result<(), String> {
156        if self.engines.iter().any(|engine| engine.id == id) {
157            return Err(format!("{id} is already configured"));
158        }
159        let Some(builtin) = builtin_by_id(id) else {
160            return Err(format!("Unknown built-in search engine {id}"));
161        };
162        self.engines.push(SearchEngine {
163            id: builtin.id.to_owned(),
164            name: builtin.name.to_owned(),
165            url: builtin.url.to_owned(),
166            alias: None,
167            order: None,
168            builtin: true,
169        });
170        Ok(())
171    }
172
173    pub(crate) fn add_user(&mut self, name: &str, url: &str) -> Result<(), String> {
174        let name = name.trim();
175        if name.is_empty() {
176            return Err("Search engine name is required".to_owned());
177        }
178        let Some(url) = normalize_search_url(url) else {
179            return Err("Search URL must contain %s or {searchTerms}".to_owned());
180        };
181        let id = unique_engine_id(self, &slug(name));
182        self.engines.push(SearchEngine {
183            id,
184            name: name.to_owned(),
185            url,
186            alias: None,
187            order: None,
188            builtin: false,
189        });
190        Ok(())
191    }
192}

201/// Remember the engines loaded at startup so `servo:settings#search` and the
202/// location bar share one list.
203pub(crate) fn install_search_engines(config_dir: Option<PathBuf>, engines: SearchEngines) {
204    *INSTALLED
205        .lock()
206        .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(InstalledSearchEngines {
207        config_dir,
208        engines,
209    });
210}
211
212/// URL template of the default engine, once [`install_search_engines`] has run.
213pub(crate) fn default_search_url() -> Option<String> {
214    INSTALLED
215        .lock()
216        .unwrap_or_else(|poisoned| poisoned.into_inner())
217        .as_ref()
218        .map(|state| state.engines.default_engine().url.clone())
219}

221/// Answer `servo:search-engines`. A query string applies one change and saves it.
222pub(crate) fn handle_search_engines_request(query: Option<&str>) -> String {
223    let mut installed = INSTALLED
224        .lock()
225        .unwrap_or_else(|poisoned| poisoned.into_inner());
226    let Some(state) = installed.as_mut() else {
227        return api_json(&SearchEngines::builtin(), None);
228    };
229
230    let mut error = None;
231    if let Some(query) = query.filter(|query| !query.is_empty()) {
232        let params: Vec<(String, String)> = url::form_urlencoded::parse(query.as_bytes())
233            .into_owned()
234            .collect();
235        let mut updated = state.engines.clone();
236        let result = apply_search_change(&mut updated, &params);
237        match result {
238            Ok(()) => match state
239                .config_dir
240                .as_deref()
241                .map(|dir| save_search_engines(dir, &updated))
242                .transpose()
243            {
244                Ok(_) => state.engines = updated,
245                Err(save_error) => error = Some(save_error),
246            },
247            Err(change_error) => error = Some(change_error),
248        }
249    }
250
251    api_json(&state.engines, error.as_deref())
252}

254/// Load `search.json` from `config_dir`.
255///
256/// A missing directory, a missing file, or a file that cannot be used falls
257/// back to [`SearchEngines::builtin`].
258pub(crate) fn load_search_engines(config_dir: Option<&Path>) -> SearchEngines {
259    let Some(config_dir) = config_dir else {
260        return SearchEngines::builtin();
261    };
262    let path = config_dir.join(SEARCH_SETTINGS_FILE);
263    if !path.exists() {
264        return SearchEngines::builtin();
265    }
266
267    let text = match fs::read_to_string(&path) {
268        Ok(text) => text,
269        Err(error) => {
270            log::warn!("Could not read {}: {error}", path.display());
271            return SearchEngines::builtin();
272        },
273    };
274    match parse_search_settings(&text) {
275        Ok(engines) => engines,
276        Err(error) => {
277            log::warn!(
278                "Could not parse {}: {error}. Using built-in search engines.",
279                path.display()
280            );
281            SearchEngines::builtin()
282        },
283    }
284}

286/// Parse a `search.json` document.
287///
288/// ```json
289/// {
290///   "version": 1,
291///   "engines": [
292///     { "id": "ddg", "order": 1 },
293///     { "id": "google", "order": 2 },
294///     {
295///       "id": "ecosia",
296///       "name": "Ecosia",
297///       "url": "<https://www.ecosia.org/search?q=%s>"
298///     }
299///   ],
300///   "defaultEngineId": "",
301///   "appDefaultEngineId": "ddg",
302///   "useSavedOrder": false
303/// }
304/// ```
305///
306/// An empty `defaultEngineId` means "use `appDefaultEngineId`". A built-in id
307/// with no `url` uses the application template. A `url` makes the entry a
308/// user engine; `{searchTerms}` is rewritten to `%s`.
309pub(crate) fn parse_search_settings(json: &str) -> Result<SearchEngines, String> {
310    let value: Value = serde_json::from_str(json).map_err(|error| error.to_string())?;
311    let Some(object) = value.as_object() else {
312        return Err("search settings must be a JSON object".to_owned());
313    };
314    let Some(entries) = object.get("engines").and_then(Value::as_array) else {
315        return Err("search settings have no engines array".to_owned());
316    };
317
318    let mut engines: Vec<SearchEngine> = Vec::new();
319    for entry in entries {
320        if let Some(engine) = engine_from_value(entry) {
321            if engines
322                .iter()
323                .any(|existing: &SearchEngine| existing.id == engine.id)
324            {
325                log::warn!("Ignoring duplicate search engine {}", engine.id);
326                continue;
327            }
328            engines.push(engine);
329        }
330    }
331    if engines.is_empty() {
332        return Err("search settings contain no usable engines".to_owned());
333    }
334
335    let use_saved_order = object
336        .get("useSavedOrder")
337        .and_then(Value::as_bool)
338        .unwrap_or(false);
339    if use_saved_order {
340        engines.sort_by(|left, right| match (left.order, right.order) {
341            (Some(left), Some(right)) => left.cmp(&right),
342            (Some(_), None) => Ordering::Less,
343            (None, Some(_)) => Ordering::Greater,
344            (None, None) => Ordering::Equal,
345        });
346    }
347
348    let default_engine_id = object
349        .get("defaultEngineId")
350        .and_then(Value::as_str)
351        .unwrap_or("");
352    let app_default_engine_id = object
353        .get("appDefaultEngineId")
354        .and_then(Value::as_str)
355        .filter(|id| !id.is_empty());
356    let default_id = select_default(&engines, default_engine_id, app_default_engine_id);
357
358    Ok(SearchEngines {
359        engines,
360        default_id,
361    })
362}

364fn engine_from_value(value: &Value) -> Option<SearchEngine> {
365    let id = string_field(value, "id").filter(|id| !id.is_empty())?;
366    let order = value
367        .get("order")
368        .and_then(Value::as_u64)
369        .and_then(|order| u32::try_from(order).ok());
370    let alias = string_field(value, "alias").filter(|alias| !alias.is_empty());
371    let name_override = string_field(value, "name").filter(|name| !name.is_empty());
372
373    if let Some(url) = string_field(value, "url") {
374        let Some(url) = normalize_search_url(&url) else {
375            log::warn!("Ignoring search engine {id}: url has no %s or {{searchTerms}} placeholder");
376            return None;
377        };
378        let name = name_override
379            .or_else(|| builtin_by_id(&id).map(|engine| engine.name.to_owned()))
380            .unwrap_or_else(|| id.clone());
381        return Some(SearchEngine {
382            id,
383            name,
384            url,
385            alias,
386            order,
387            builtin: false,
388        });
389    }
390
391    let Some(builtin) = builtin_by_id(&id) else {
392        log::warn!("Ignoring search engine {id}: unknown built-in id and no url");
393        return None;
394    };
395    Some(SearchEngine {
396        id,
397        name: name_override.unwrap_or_else(|| builtin.name.to_owned()),
398        url: builtin.url.to_owned(),
399        alias,
400        order,
401        builtin: true,
402    })
403}

405fn select_default(
406    engines: &[SearchEngine],
407    default_engine_id: &str,
408    app_default_engine_id: Option<&str>,
409) -> String {
410    if !default_engine_id.is_empty() && engines.iter().any(|engine| engine.id == default_engine_id)
411    {
412        return default_engine_id.to_owned();
413    }
414    if let Some(app_default) = app_default_engine_id &&
415        engines.iter().any(|engine| engine.id == app_default)
416    {
417        return app_default.to_owned();
418    }
419    if engines
420        .iter()
421        .any(|engine| engine.id == APP_DEFAULT_ENGINE_ID)
422    {
423        return APP_DEFAULT_ENGINE_ID.to_owned();
424    }
425    engines[0].id.clone()
426}

436fn apply_search_change(
437    engines: &mut SearchEngines,
438    params: &[(String, String)],
439) -> Result<(), String> {
440    let value = |key: &str| {
441        params
442            .iter()
443            .find(|(name, _)| name == key)
444            .map(|(_, value)| value.as_str())
445    };
446    if let Some(id) = value("default") {
447        engines.set_default(id)
448    } else if let Some(id) = value("remove") {
449        engines.remove(id)
450    } else if let Some(id) = value("addBuiltin") {
451        engines.add_builtin(id)
452    } else if value("name").is_some() || value("url").is_some() {
453        engines.add_user(value("name").unwrap_or(""), value("url").unwrap_or(""))
454    } else {
455        Err("Unknown search settings request".to_owned())
456    }
457}

459fn save_search_engines(config_dir: &Path, engines: &SearchEngines) -> Result<(), String> {
460    fs::create_dir_all(config_dir).map_err(|error| error.to_string())?;
461    let text = serde_json::to_string_pretty(&settings_value(engines))
462        .map_err(|error| error.to_string())?;
463    fs::write(config_dir.join(SEARCH_SETTINGS_FILE), format!("{text}\n"))
464        .map_err(|error| error.to_string())
465}

467fn settings_value(engines: &SearchEngines) -> Value {
468    let entries: Vec<Value> = engines
469        .engines
470        .iter()
471        .enumerate()
472        .map(|(index, engine)| {
473            let mut entry = serde_json::Map::new();
474            entry.insert("id".into(), Value::String(engine.id.clone()));
475            entry.insert("name".into(), Value::String(engine.name.clone()));
476            entry.insert("order".into(), Value::from(index as u64 + 1));
477            if let Some(alias) = &engine.alias {
478                entry.insert("alias".into(), Value::String(alias.clone()));
479            }
480            if !engine.builtin {
481                entry.insert("url".into(), Value::String(engine.url.clone()));
482            }
483            Value::Object(entry)
484        })
485        .collect();
486    serde_json::json!({
487        "version": 1,
488        "useSavedOrder": true,
489        "appDefaultEngineId": APP_DEFAULT_ENGINE_ID,
490        "defaultEngineId": engines.default_id,
491        "engines": entries,
492    })
493}

495fn api_json(engines: &SearchEngines, error: Option<&str>) -> String {
496    let configured = engines
497        .engines
498        .iter()
499        .map(|engine| {
500            serde_json::json!({
501                "id": engine.id,
502                "name": engine.name,
503                "url": engine.url,
504                "builtin": engine.builtin,
505            })
506        })
507        .collect::<Vec<_>>();
508    let available = BUILTIN_ENGINES
509        .iter()
510        .filter(|builtin| !engines.engines.iter().any(|engine| engine.id == builtin.id))
511        .map(|builtin| serde_json::json!({ "id": builtin.id, "name": builtin.name }))
512        .collect::<Vec<_>>();
513    let mut value = serde_json::json!({
514        "defaultId": engines.default_id,
515        "engines": configured,
516        "available": available,
517    });
518    if let Some(error) = error {
519        value["error"] = Value::String(error.to_owned());
520    }
521    serde_json::to_string(&value)
522        .unwrap_or_else(|_| "{\"error\":\"Could not encode search engines\"}".to_owned())
523}

525fn slug(name: &str) -> String {
526    let mut id = String::new();
527    for character in name.chars() {
528        if character.is_ascii_alphanumeric() {
529            id.push(character.to_ascii_lowercase());
530        } else if !id.is_empty() && !id.ends_with('-') {
531            id.push('-');
532        }
533    }
534    let id = id.trim_matches('-');
535    if id.is_empty() {
536        "engine".to_owned()
537    } else {
538        id.to_owned()
539    }
540}

542fn unique_engine_id(engines: &SearchEngines, base: &str) -> String {
543    if !engines.engines.iter().any(|engine| engine.id == base) {
544        return base.to_owned();
545    }
546    let mut suffix = 2u32;
547    loop {
548        let candidate = format!("{base}-{suffix}");
549        if !engines.engines.iter().any(|engine| engine.id == candidate) {
550            return candidate;
551        }
552        suffix += 1;
553    }
554}

556/// Accept Aether's `%s` and Firefox's `{searchTerms}`.
557fn normalize_search_url(url: &str) -> Option<String> {
558    let url = url.trim();
559    if url.is_empty() {
560        return None;
561    }
562    let normalized = url.replace("{searchTerms}", "%s");
563    if !normalized.contains("%s") {
564        return None;
565    }
566    let parsed = url::Url::parse(&normalized.replace("%s", "test")).ok()?;
567    matches!(parsed.scheme(), "http" | "https").then_some(normalized)
568}
