5//! SVG toolbar icons. Add new buttons by dropping an SVG in `resources/icons/`
6//! and a variant on [`ToolbarIcon`].

8use std::collections::HashMap;
9
10use egui::load::SizedTexture;
11use log::warn;
12use resvg::{tiny_skia, usvg};

14/// Logical size of a toolbar icon, in egui points.
15const TOOLBAR_ICON_SIZE: f32 = 14.0;
16
17/// Toolbar button hit target.
18const TOOLBAR_BUTTON_SIZE: f32 = 20.0;
19
20/// Rounded square used to mark a toolbar button on hover and press.
21const TOOLBAR_BUTTON_CORNER_RADIUS: u8 = 4;
22
23/// Toolbar button: icon only at rest, rounded square on hover, darker on press.
24pub(crate) fn add_toolbar_button(ui: &mut egui::Ui, button: egui::Button<'_>) -> egui::Response {
25    // Reserve a paint slot behind the icon so hover/press fill never covers it.
26    let background = ui.painter().add(egui::Shape::Noop);
27    let response = ui.add(
28        button
29            .frame(false)
30            .min_size(egui::vec2(TOOLBAR_BUTTON_SIZE, TOOLBAR_BUTTON_SIZE)),
31    );
32    if let Some(fill) = toolbar_button_fill_color(
33        ui.visuals(),
34        toolbar_button_visual_state(
35            ui.is_enabled(),
36            response.hovered(),
37            response.is_pointer_button_down_on(),
38        ),
39    ) {
40        ui.painter().set(
41            background,
42            egui::Shape::rect_filled(
43                response.rect,
44                egui::CornerRadius::same(TOOLBAR_BUTTON_CORNER_RADIUS),
45                fill,
46            ),
47        );
48    }
49    response
50}

52#[derive(Clone, Copy, Debug, Eq, PartialEq)]
53enum ToolbarButtonVisualState {
54    Rest,
55    Hover,
56    Press,
57}
58
59fn toolbar_button_visual_state(
60    enabled: bool,
61    hovered: bool,
62    pressed: bool,
63) -> ToolbarButtonVisualState {
64    if enabled && pressed {
65        ToolbarButtonVisualState::Press
66    } else if enabled && hovered {
67        ToolbarButtonVisualState::Hover
68    } else {
69        ToolbarButtonVisualState::Rest
70    }
71}
72
73fn toolbar_button_fill_color(
74    visuals: &egui::Visuals,
75    state: ToolbarButtonVisualState,
76) -> Option<egui::Color32> {
77    match state {
78        ToolbarButtonVisualState::Rest => None,
79        ToolbarButtonVisualState::Hover => Some(visuals.widgets.hovered.weak_bg_fill),
80        ToolbarButtonVisualState::Press => Some(visuals.widgets.active.weak_bg_fill),
81    }
82}

131/// Rasterized SVG textures, cached by icon and physical pixel size.
132#[derive(Default)]
133pub(crate) struct ToolbarIconCache {
134    textures: HashMap<(ToolbarIcon, u32), egui::TextureHandle>,
135}
136
137impl ToolbarIconCache {
138    pub(crate) fn button(&mut self, ui: &mut egui::Ui, icon: ToolbarIcon) -> egui::Response {
139        let button = self.image_button(ui, icon);
140        add_toolbar_button(ui, button)
141    }
142
143    /// Icon button without the toolbar hover square, for chrome that paints its own background.
144    pub(crate) fn image_button(&mut self, ui: &egui::Ui, icon: ToolbarIcon) -> egui::Button<'_> {
145        let button = match self.image(ui, icon) {
146            Some(image) => egui::Button::new(image),
147            None => egui::Button::new(""),
148        };
149        button.image_tint_follows_text_color(true)
150    }
151
152    /// A tintable icon image for embedding in another widget, such as a menu row.
153    pub(crate) fn image(
154        &mut self,
155        ui: &egui::Ui,
156        icon: ToolbarIcon,
157    ) -> Option<egui::Image<'static>> {
158        let pixel_size = (TOOLBAR_ICON_SIZE * ui.pixels_per_point()).round().max(1.0) as u32;
159        let handle = self.texture(ui.ctx(), icon, pixel_size)?;
160        Some(
161            egui::Image::from_texture(SizedTexture::new(
162                handle.id(),
163                egui::vec2(TOOLBAR_ICON_SIZE, TOOLBAR_ICON_SIZE),
164            ))
165            .fit_to_exact_size(egui::vec2(TOOLBAR_ICON_SIZE, TOOLBAR_ICON_SIZE)),
166        )
167    }
168
169    fn texture(
170        &mut self,
171        ctx: &egui::Context,
172        icon: ToolbarIcon,
173        pixel_size: u32,
174    ) -> Option<egui::TextureHandle> {
175        if let Some(handle) = self.textures.get(&(icon, pixel_size)) {
176            return Some(handle.clone());
177        }
178
179        let color_image = rasterize_svg(icon.svg_bytes(), pixel_size)?;
180        let handle = ctx.load_texture(
181            format!("{}@{pixel_size}", icon.texture_name()),
182            color_image,
183            egui::TextureOptions::LINEAR,
184        );
185        self.textures.insert((icon, pixel_size), handle.clone());
186        Some(handle)
187    }
188}

190/// SVGs use `context-fill` / `context-fill-opacity`, which resvg
191/// treats as an invalid paint (fully transparent). Map them to an opaque white
192/// fill so egui can tint the icon to the current text color.
193fn prepare_toolbar_svg(svg: &[u8]) -> std::borrow::Cow<'_, [u8]> {
194    let Ok(text) = std::str::from_utf8(svg) else {
195        return std::borrow::Cow::Borrowed(svg);
196    };
197    if !text.contains("context-fill") {
198        return std::borrow::Cow::Borrowed(svg);
199    }
200    std::borrow::Cow::Owned(
201        text.replace("context-fill-opacity", "1")
202            .replace("context-fill", "#ffffff")
203            .into_bytes(),
204    )
205}
206
207pub(crate) fn rasterize_svg(svg: &[u8], size_px: u32) -> Option<egui::ColorImage> {
208    let svg = prepare_toolbar_svg(svg);
209    let tree = usvg::Tree::from_data(&svg, &usvg::Options::default())
210        .map_err(|error| {
211            warn!("Failed to parse toolbar SVG: {error}");
212            error
213        })
214        .ok()?;
215    let mut pixmap = tiny_skia::Pixmap::new(size_px, size_px)?;
216    let svg_size = tree.size();
217    let transform = tiny_skia::Transform::from_scale(
218        size_px as f32 / svg_size.width(),
219        size_px as f32 / svg_size.height(),
220    );
221    resvg::render(&tree, transform, &mut pixmap.as_mut());
222    Some(egui::ColorImage::from_rgba_premultiplied(
223        [size_px as usize, size_px as usize],
224        pixmap.data(),
225    ))
226}
