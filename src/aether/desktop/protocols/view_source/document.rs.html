5//! Builds the numbered listing shown for a `view-source:` URL.
6
7use super::markup::{self, Token, TokenKind};
8
9/// Inlined rather than linked, so the listing pulls in no subresources.
10const STYLE: &str = include_str!("../../../../../resources/view-source.css");
11
12/// Servo has no CSS counters, so line numbers are written into the markup. They live
13/// in their own element so that the stylesheet can set them apart from the source.
14pub(super) fn source_document(viewed_url: &str, source: &str, highlight: bool) -> String {
15    let title = escaped(viewed_url);
16    let line_count = source.split('\n').count().max(1);
17    let number_width = line_count.to_string().len().max(2);
18    let lines = lines(source, highlight);
19    format!(
20        "<!DOCTYPE html>\n<html>\n<head>\n<meta charset=\"utf-8\">\n\
21         <title>Source of {title}</title>\n\
22         <style>\n{STYLE}</style>\n</head>\n\
23         <body><pre id=\"source\" style=\"--line-number-width: {number_width}ch\">\
24         {lines}</pre></body>\n</html>\n"
25    )
26}
27
28/// One block per source line. Nothing separates the blocks, because a newline inside
29/// the preformatted listing would show up as a blank line of its own.
30fn lines(source: &str, highlight: bool) -> String {
31    let tokens = if highlight {
32        markup::tokenize(source)
33    } else {
34        vec![Token {
35            kind: TokenKind::Text,
36            text: source,
37        }]
38    };
39
40    let mut lines = String::new();
41    let mut number = 1;
42    open_line(&mut lines, number);
43
44    for token in tokens {
45        for (index, part) in token.text.split('\n').enumerate() {
46            if index > 0 {
47                lines.push_str("</span>");
48                number += 1;
49                open_line(&mut lines, number);
50            }
51            // The parser turns a carriage return back into a line break, which would
52            // split CRLF source across two lines.
53            push_run(
54                &mut lines,
55                part.strip_suffix('\r').unwrap_or(part),
56                token.kind,
57            );
58        }
59    }
60
61    lines.push_str("</span>");
62    lines
63}
64
65fn open_line(out: &mut String, number: usize) {
66    out.push_str("<span class=\"line\"><span class=\"line-number\">");
67    out.push_str(&number.to_string());
68    out.push_str("</span>");
69}
70
71fn push_run(out: &mut String, text: &str, kind: TokenKind) {
72    if text.is_empty() {
73        return;
74    }
75    match kind.class() {
76        Some(class) => {
77            out.push_str("<span class=\"");
78            out.push_str(class);
79            out.push_str("\">");
80            push_escaped(out, text);
81            out.push_str("</span>");
82        },
83        None => push_escaped(out, text),
84    }
85}
86
87fn escaped(text: &str) -> String {
88    let mut escaped = String::with_capacity(text.len());
89    push_escaped(&mut escaped, text);
90    escaped
91}
92
93fn push_escaped(out: &mut String, text: &str) {
94    for character in text.chars() {
95        match character {
96            '&' => out.push_str("&amp;"),
97            '<' => out.push_str("&lt;"),
98            '>' => out.push_str("&gt;"),
99            '"' => out.push_str("&quot;"),
100            _ => out.push(character),
101        }
102    }
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108
109    #[test]
110    fn every_source_line_is_numbered() {
111        let lines = lines("one\ntwo\n\nfour", false);
112        for number in 1..=4 {
113            assert!(
114                lines.contains(&format!("<span class=\"line-number\">{number}</span>")),
115                "line {number} should be numbered in {lines}"
116            );
117        }
118        assert_eq!(lines.matches("class=\"line\"").count(), 4);
119    }
120
121    /// A trailing newline ends a line, so `"one\n"` is two line blocks (the second empty).
122    #[test]
123    fn a_trailing_newline_adds_an_empty_line() {
124        assert_eq!(lines("one\n", false).matches("class=\"line\"").count(), 2);
125        assert_eq!(lines("one", false).matches("class=\"line\"").count(), 1);
126    }
127
128    #[test]
129    fn crlf_source_does_not_double_space() {
130        let lines = lines("one\r\ntwo", false);
131        assert!(!lines.contains('\r'), "{lines}");
132        assert_eq!(lines.matches("class=\"line\"").count(), 2);
133    }
134
135    #[test]
136    fn markup_is_escaped_and_coloured() {
137        let lines = lines("<p class=\"a\">x &amp; y</p>", true);
138        assert!(
139            lines.contains("<span class=\"tag\">&lt;p</span>"),
140            "{lines}"
141        );
142        assert!(
143            lines.contains("<span class=\"attribute-name\">class</span>"),
144            "{lines}"
145        );
146        assert!(
147            lines.contains("<span class=\"entity\">&amp;amp;</span>"),
148            "{lines}"
149        );
150        assert!(
151            lines.contains("<span class=\"attribute-value\">&quot;a&quot;</span>"),
152            "{lines}"
153        );
154    }
155
156    #[test]
157    fn unhighlighted_source_is_still_escaped() {
158        let lines = lines("<p>", false);
159        assert!(lines.contains("&lt;p&gt;"), "{lines}");
160        assert!(!lines.contains("class=\"tag\""), "{lines}");
161    }
162
163    #[test]
164    fn a_run_spanning_lines_is_numbered_on_each_line() {
165        let lines = lines("<!-- a\nb -->", true);
166        assert_eq!(lines.matches("class=\"line\"").count(), 2);
167        assert_eq!(lines.matches("class=\"comment\"").count(), 2);
168    }
169
170    #[test]
171    fn the_title_names_the_viewed_url() {
172        let document = source_document("https://example.com/?a=1&b=2", "x", false);
173        assert!(
174            document.contains("<title>Source of https://example.com/?a=1&amp;b=2</title>"),
175            "{document}"
176        );
177    }
178
179    #[test]
180    fn the_gutter_grows_with_the_line_count() {
181        let ten = source_document("u", &"x\n".repeat(9), false);
182        assert!(
183            ten.contains("--line-number-width: 2ch"),
184            "ten lines should keep the minimum two-digit gutter: {ten}"
185        );
186
187        let thousand = source_document("u", &"x\n".repeat(999), false);
188        assert!(
189            thousand.contains("--line-number-width: 4ch"),
190            "1000 lines need a four-digit gutter: {thousand}"
191        );
192    }
