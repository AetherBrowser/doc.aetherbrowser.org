5//! Profile files for the network filter edited from `about:settings#network`.
6//!
7//! The network stack reads the same files (`blocklist.txt` and
8//! `content_filter.json`) and applies them on the next request.
9//!
10//! The JSON keys are the Firefox preference names, as in `home.json`:
11//! - `privacy.trackingprotection.enabled`
12//! - `urlclassifier.trackingWhitelistTable` — hosts that are never blocked
13//! - `urlclassifier.trackingTable` — extra filter files, beside `blocklist.txt`
14
15use std::fs;
16use std::path::{Path, PathBuf};
17use std::sync::Mutex;
18
19use serde_json::{Value, json};
20
21const BLOCKLIST_FILE: &str = "blocklist.txt";
22const CONFIG_FILE: &str = "content_filter.json";
23
24const PREF_ENABLED: &str = "privacy.trackingprotection.enabled";
25const PREF_WHITELIST: &str = "urlclassifier.trackingWhitelistTable";
26const PREF_FILTER_LISTS: &str = "urlclassifier.trackingTable";
27
28struct InstalledFilter {
29    config_dir: Option<PathBuf>,
30}
31
32static INSTALLED: Mutex<InstalledFilter> = Mutex::new(InstalledFilter { config_dir: None });
33
34/// Remember the profile directory so `about:content-filter` writes the list
35/// the network stack is watching.
36pub(crate) fn install_content_filter(config_dir: Option<PathBuf>) {
37    INSTALLED
38        .lock()
39        .unwrap_or_else(|poisoned| poisoned.into_inner())
40        .config_dir = config_dir;
41}
42
43/// Answer `about:content-filter`. A query string saves the enable flag and the list.
44pub(crate) fn handle_content_filter_request(query: Option<&str>) -> String {
45    let config_dir = INSTALLED
46        .lock()
47        .unwrap_or_else(|poisoned| poisoned.into_inner())
48        .config_dir
49        .clone();
50    let Some(config_dir) = config_dir else {
51        return json_settings(true, "", Some("No profile directory is available."));
52    };
53
54    if let Some(query) = query.filter(|query| !query.is_empty()) {
55        let params: Vec<(String, String)> = url::form_urlencoded::parse(query.as_bytes())
56            .into_owned()
57            .collect();
58        let current = load_settings(&config_dir);
59        let enabled = params
60            .iter()
61            .find(|(name, _)| name == "enabled")
62            .map(|(_, value)| value == "true")
63            .unwrap_or(current.enabled);
64        let blocklist = params
65            .iter()
66            .find(|(name, _)| name == "blocklist")
67            .map(|(_, value)| value.clone())
68            .unwrap_or_else(|| current.blocklist.clone());
69        if let Err(error) = save_settings(&config_dir, enabled, &blocklist) {
70            return json_settings(current.enabled, &current.blocklist, Some(&error));
71        }
72    }
73
74    let settings = load_settings(&config_dir);
75    json_settings(settings.enabled, &settings.blocklist, None)
76}
77
78struct FilterSettings {
79    enabled: bool,
80    blocklist: String,
81}
82
83fn load_settings(config_dir: &Path) -> FilterSettings {
84    let enabled = fs::read_to_string(config_dir.join(CONFIG_FILE))
85        .ok()
86        .and_then(|text| serde_json::from_str::<Value>(&text).ok())
87        .and_then(|value| {
88            value
89                .get(PREF_ENABLED)
90                .or_else(|| value.get("enabled"))
91                .and_then(Value::as_bool)
92        })
93        .unwrap_or(true);
94    let blocklist = fs::read_to_string(config_dir.join(BLOCKLIST_FILE)).unwrap_or_default();
95    FilterSettings { enabled, blocklist }
96}
97
98fn save_settings(config_dir: &Path, enabled: bool, blocklist: &str) -> Result<(), String> {
99    fs::create_dir_all(config_dir).map_err(|error| error.to_string())?;
100    let current = fs::read_to_string(config_dir.join(CONFIG_FILE))
101        .ok()
102        .and_then(|text| serde_json::from_str::<Value>(&text).ok())
103        .unwrap_or_else(|| json!({}));
104    let config = json!({
105        PREF_ENABLED: enabled,
106        PREF_WHITELIST: string_list(
107            current
108                .get(PREF_WHITELIST)
109                .or_else(|| current.get("whitelist_domains")),
110        ),
111        PREF_FILTER_LISTS: string_list(
112            current
113                .get(PREF_FILTER_LISTS)
114                .or_else(|| current.get("filter_lists")),
115        ),
116    });
117    let config_text = serde_json::to_string_pretty(&config).map_err(|error| error.to_string())?;
118    let config_bytes = format!("{config_text}\n");
119    let mut list = blocklist.replace("\r\n", "\n");
120    if !list.is_empty() && !list.ends_with('\n') {
121        list.push('\n');
122    }
123    let list_path = config_dir.join(BLOCKLIST_FILE);
124    let config_path = config_dir.join(CONFIG_FILE);
125    // Enabling must not become visible before the list it belongs with. A
126    // failed second write then leaves the filter off, or already off, instead
127    // of matching the previous list under the new flag.
128    if enabled {
129        write_atomic(&list_path, list.as_bytes())?;
130        write_atomic(&config_path, config_bytes.as_bytes())
131    } else {
132        write_atomic(&config_path, config_bytes.as_bytes())?;
133        write_atomic(&list_path, list.as_bytes())
134    }
135}
136
137fn write_atomic(path: &Path, bytes: &[u8]) -> Result<(), String> {
138    let temporary = path.with_extension("tmp");
139    fs::write(&temporary, bytes).map_err(|error| error.to_string())?;
140    fs::rename(&temporary, path).map_err(|error| error.to_string())
141}
142
143fn string_list(value: Option<&Value>) -> Vec<String> {
144    value
145        .and_then(Value::as_array)
146        .map(|items| {
147            items
148                .iter()
149                .filter_map(|item| item.as_str().map(str::to_owned))
150                .collect()
151        })
152        .unwrap_or_default()
153}
154
155fn json_settings(enabled: bool, blocklist: &str, error: Option<&str>) -> String {
156    let mut value = json!({
157        "enabled": enabled,
158        "blocklist": blocklist,
159    });
160    if let Some(error) = error {
161        value["error"] = Value::String(error.to_owned());
162    }
163    value.to_string()
164}

166#[cfg(test)]
167mod tests {
168    use super::*;
169
170    fn temp_dir() -> PathBuf {
171        let nanos = std::time::SystemTime::now()
172            .duration_since(std::time::UNIX_EPOCH)
173            .unwrap()
174            .as_nanos();
175        let dir = std::env::temp_dir().join(format!(
176            "aether-filter-settings-{}-{nanos}",
177            std::process::id()
178        ));
179        fs::create_dir_all(&dir).unwrap();
180        dir
181    }
182
183    #[test]
184    fn saving_the_list_preserves_extra_filter_files() {
185        let dir = temp_dir();
186        fs::write(
187            dir.join(CONFIG_FILE),
188            "{ \"enabled\": true, \"filter_lists\": [\"easylist.txt\"], \"whitelist_domains\": [\"ok.example\"] }\n",
189        )
190        .unwrap();
191
192        save_settings(&dir, true, "ads.example\nhttps://tracker.example/pixel\n").unwrap();
193
194        let stored: Value =
195            serde_json::from_str(&fs::read_to_string(dir.join(CONFIG_FILE)).unwrap()).unwrap();
196        assert_eq!(stored["privacy.trackingprotection.enabled"], true);
197        assert_eq!(stored["urlclassifier.trackingTable"][0], "easylist.txt");
198        assert_eq!(
199            stored["urlclassifier.trackingWhitelistTable"][0],
200            "ok.example"
201        );
202        assert!(stored.get("enabled").is_none());
203        assert_eq!(
204            fs::read_to_string(dir.join(BLOCKLIST_FILE)).unwrap(),
205            "ads.example\nhttps://tracker.example/pixel\n"
206        );
207
208        let loaded = load_settings(&dir);
209        assert!(loaded.enabled);
210        assert!(loaded.blocklist.contains("ads.example"));
211        let _ = fs::remove_dir_all(&dir);
212    }
213
214    #[test]
215    fn a_request_updates_the_installed_profile() {
216        let dir = temp_dir();
217        install_content_filter(Some(dir.clone()));
218        let body = handle_content_filter_request(Some("enabled=false&blocklist=blocked.example"));
219        let value: Value = serde_json::from_str(&body).unwrap();
220        assert_eq!(value["enabled"], false);
221        assert_eq!(value["blocklist"], "blocked.example\n");
222        assert!(value.get("error").is_none());
223
224        let again: Value = serde_json::from_str(&handle_content_filter_request(None)).unwrap();
225        assert_eq!(again["enabled"], false);
226        assert_eq!(again["blocklist"], "blocked.example\n");
227        let _ = fs::remove_dir_all(&dir);
228        install_content_filter(None);
229    }
