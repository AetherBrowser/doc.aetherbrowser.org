12/// Schema migrations, in order. The database's `user_version` is the number of migrations
13/// already applied, so existing entries must never change: append a new one instead.
14const MIGRATIONS: &[&str] = &[
15    // Version 1: URLs and their visits. `places` will also be referenced by bookmarks, and its
16    // `guid` identifies a URL across devices for a future sync.
17    "CREATE TABLE places (
18        id INTEGER PRIMARY KEY,
19        url TEXT NOT NULL UNIQUE,
20        guid TEXT NOT NULL UNIQUE
21    );
22    CREATE TABLE visits (
23        id INTEGER PRIMARY KEY,
24        place_id INTEGER NOT NULL REFERENCES places(id) ON DELETE CASCADE,
25        visit_date INTEGER NOT NULL
26    );
27    CREATE INDEX visits_place_id_index ON visits(place_id);
28    CREATE INDEX visits_date_index ON visits(visit_date);",
29    // Version 2: the last title of each page, as shown in its tab.
30    "ALTER TABLE places ADD COLUMN title TEXT;",
31];
32
33pub(crate) enum HistoryCommand {
34    /// Records a visit of `url` at `visit_date`.
35    RecordVisit {
36        url: Url,
37        visit_date: SystemTime,
38        title: Option<String>,
39    },
40    /// Stores `title` as the last title of `url`, if `url` was visited.
41    SetTitle { url: Url, title: String },
42    /// Moves the visit of `old_url` at `visit_date` to `new_url`, or deletes it when `new_url`
43    /// is `None`.
44    ReplaceVisit {
45        old_url: Url,
46        visit_date: SystemTime,
47        new_url: Option<Url>,
48        title: Option<String>,
49    },
50}

52pub struct HistoryStore {
53    connection: Connection,
54}

56impl HistoryStore {
57    /// Opens the history database at `path`, creating it if needed.
58    pub fn open(path: &Path) -> rusqlite::Result<Self> {
59        let connection = Connection::open(path)?;
60        connection.pragma_update(None, "journal_mode", "WAL")?;
61        connection.pragma_update(None, "synchronous", "NORMAL")?;
62        Self::new(connection)
63    }
64
65    /// Opens a history that is never written to disk.
66    pub fn open_in_memory() -> rusqlite::Result<Self> {
67        Self::new(Connection::open_in_memory()?)
68    }
69
70    fn new(mut connection: Connection) -> rusqlite::Result<Self> {
71        connection.pragma_update(None, "foreign_keys", true)?;
72        connection.pragma_update(None, "secure_delete", true)?;
73        migrate(&mut connection)?;
74        Ok(Self { connection })
75    }
76
77    /// Writes `commands` in order, in one transaction. A title that a later command replaces
78    /// is not written.
79    pub(crate) fn apply(&mut self, commands: &[HistoryCommand]) -> rusqlite::Result<()> {
80        let mut last_titles = HashMap::new();
81        for (index, command) in commands.iter().enumerate() {
82            if let HistoryCommand::SetTitle { url, .. } = command {
83                last_titles.insert(url, index);
84            }
85        }
86        let transaction = self.connection.transaction()?;
87        for (index, command) in commands.iter().enumerate() {
88            match command {
89                HistoryCommand::RecordVisit {
90                    url,
91                    visit_date,
92                    title,
93                } => record_visit(&transaction, url, *visit_date, title.as_deref())?,
94                HistoryCommand::SetTitle { url, title } => {
95                    if last_titles[url] == index {
96                        set_title(&transaction, url, title)?
97                    }
98                },
99                HistoryCommand::ReplaceVisit {
100                    old_url,
101                    visit_date,
102                    new_url,
103                    title,
104                } => replace_visit(
105                    &transaction,
106                    old_url,
107                    *visit_date,
108                    new_url.as_ref(),
109                    title.as_deref(),
110                )?,
111            }
112        }
113        transaction.commit()
114    }
115
116    /// Deletes the visits before `date`, and the places left without visits.
117    pub(crate) fn delete_visits_before(&mut self, date: SystemTime) -> rusqlite::Result<()> {
118        self.delete_visits("DELETE FROM visits WHERE visit_date < ?1", date)
119    }
120
121    /// Deletes the visits at or after `date`, and the places left without visits.
122    pub(crate) fn delete_visits_since(&mut self, date: SystemTime) -> rusqlite::Result<()> {
123        self.delete_visits("DELETE FROM visits WHERE visit_date >= ?1", date)
124    }
125
126    /// Also erases the deleted rows from the database files, so that they cannot be read back.
127    fn delete_visits(&mut self, statement: &str, date: SystemTime) -> rusqlite::Result<()> {
128        let transaction = self.connection.transaction()?;
129        let deleted = transaction.execute(statement, [microseconds_since_epoch(date)])?;
130        if deleted == 0 {
131            return Ok(());
132        }
133        transaction.execute(
134            "DELETE FROM places
135             WHERE NOT EXISTS (SELECT 1 FROM visits WHERE visits.place_id = places.id)",
136            [],
137        )?;
138        transaction.commit()?;
139        self.connection
140            .query_row("PRAGMA wal_checkpoint(TRUNCATE)", [], |_| Ok(()))
141    }
142}

144fn record_visit(
145    connection: &Connection,
146    url: &Url,
147    visit_date: SystemTime,
148    title: Option<&str>,
149) -> rusqlite::Result<()> {
150    let url = stored_url(url);
151    insert_place(connection, &url, title)?;
152    connection.execute(
153        "INSERT INTO visits (place_id, visit_date) SELECT id, ?2 FROM places WHERE url = ?1",
154        params![url, microseconds_since_epoch(visit_date)],
155    )?;
156    Ok(())
157}
158
159fn set_title(connection: &Connection, url: &Url, title: &str) -> rusqlite::Result<()> {
160    connection.execute(
161        "UPDATE places SET title = ?2 WHERE url = ?1",
162        params![stored_url(url), title],
163    )?;
164    Ok(())
165}
166
167fn replace_visit(
168    connection: &Connection,
169    old_url: &Url,
170    visit_date: SystemTime,
171    new_url: Option<&Url>,
172    title: Option<&str>,
173) -> rusqlite::Result<()> {
174    let old_url = stored_url(old_url);
175    let new_url = new_url.map(stored_url);
176    let visit_date = microseconds_since_epoch(visit_date);
177    let visit_id: Option<i64> = connection
178        .query_row(
179            "SELECT visits.id FROM visits JOIN places ON places.id = visits.place_id
180             WHERE places.url = ?1 AND visits.visit_date = ?2
181             ORDER BY visits.id DESC LIMIT 1",
182            params![old_url, visit_date],
183            |row| row.get(0),
184        )
185        .optional()?;
186    let Some(visit_id) = visit_id else {
187        return Ok(());
188    };
189    match &new_url {
190        Some(new_url) => {
191            insert_place(connection, new_url, title)?;
192            connection.execute(
193                "UPDATE visits SET place_id = (SELECT id FROM places WHERE url = ?1) WHERE id = ?2",
194                params![new_url, visit_id],
195            )?
196        },
197        None => connection.execute("DELETE FROM visits WHERE id = ?1", [visit_id])?,
198    };
199    connection.execute(
200        "DELETE FROM places WHERE url = ?1
201         AND NOT EXISTS (SELECT 1 FROM visits WHERE visits.place_id = places.id)",
202        [&old_url],
203    )?;
204    Ok(())
205}

207/// Adds `url` to the places if needed, and updates its title when `title` is known.
208fn insert_place(connection: &Connection, url: &str, title: Option<&str>) -> rusqlite::Result<()> {
209    connection.execute(
210        "INSERT INTO places (url, guid, title) VALUES (?1, lower(hex(randomblob(16))), ?2)
211         ON CONFLICT (url) DO UPDATE SET title = coalesce(excluded.title, places.title)",
212        params![url, title],
213    )?;
214    Ok(())
215}

217fn stored_url(url: &Url) -> String {
218    let mut url = url.clone();
219    let _ = url.set_username("");
220    let _ = url.set_password(None);
221    url.into()
222}
223
224fn migrate(connection: &mut Connection) -> rusqlite::Result<()> {
225    let version: u32 = connection.pragma_query_value(None, "user_version", |row| row.get(0))?;
226    if version as usize > MIGRATIONS.len() {
227        return Err(rusqlite::Error::SqliteFailure(
228            rusqlite::ffi::Error::new(rusqlite::ffi::SQLITE_ERROR),
229            Some(format!(
230                "schema version {version} is newer than the supported version {}",
231                MIGRATIONS.len()
232            )),
233        ));
234    }
235    let transaction = connection.transaction()?;
236    for (migration, new_version) in MIGRATIONS.iter().zip(1_u32..).skip(version as usize) {
237        transaction.execute_batch(migration)?;
238        transaction.pragma_update(None, "user_version", new_version)?;
239    }
240    transaction.commit()
241}

243/// Dates before the Unix epoch, which only a wrong system clock produces, are stored as 0.
244fn microseconds_since_epoch(time: SystemTime) -> i64 {
245    time.duration_since(UNIX_EPOCH)
246        .map_or(0, |duration| duration.as_micros() as i64)
247}
