14impl PlacesStore {
15    pub fn bookmarked_urls(&self) -> rusqlite::Result<Vec<Url>> {
16        let mut statement = self
17            .connection
18            .prepare("SELECT url FROM places JOIN bookmarks ON bookmarks.place_id = places.id")?;
19        let urls = statement
20            .query_map([], |row| row.get::<_, String>(0))?
21            .collect::<Result<Vec<_>, _>>()?;
22        Ok(urls.iter().filter_map(|url| Url::parse(url).ok()).collect())
23    }
24}
25
26pub(crate) fn add_bookmark(
27    connection: &Connection,
28    url: &Url,
29    title: Option<&str>,
30    date_added: SystemTime,
31) -> rusqlite::Result<()> {
32    let url = stored_url(url);
33    insert_place(connection, &url, None)?;
34    connection.execute(
35        "INSERT INTO bookmarks (place_id, title, date_added, guid)
36         SELECT id, ?2, ?3, lower(hex(randomblob(16))) FROM places WHERE url = ?1
37         ON CONFLICT (place_id) DO NOTHING",
38        params![url, title, microseconds_since_epoch(date_added)],
39    )?;
40    Ok(())
41}
42
43pub(crate) fn remove_bookmark(connection: &Connection, url: &Url) -> rusqlite::Result<()> {
44    let url = stored_url(url);
45    connection.execute(
46        "DELETE FROM bookmarks WHERE place_id = (SELECT id FROM places WHERE url = ?1)",
47        [&url],
48    )?;
49    delete_place_if_unused(connection, &url)
50}

95    fn bookmarks(connection: &Connection) -> Vec<(String, Option<String>, i64)> {
96        connection
97            .prepare(
98                "SELECT url, bookmarks.title, date_added
99                 FROM bookmarks JOIN places ON places.id = bookmarks.place_id
100                 ORDER BY bookmarks.id",
101            )
102            .unwrap()
103            .query_map([], |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)))
104            .unwrap()
105            .collect::<Result<_, _>>()
106            .unwrap()
107    }

112            .unwrap()
113            .query_map([], |row| row.get(0))
114            .unwrap()
115            .collect::<Result<_, _>>()
116            .unwrap()
117    }

119    fn visit_count(connection: &Connection) -> u32 {
120        connection
121            .query_row("SELECT count(*) FROM visits", [], |row| row.get(0))
122            .unwrap()
123    }
