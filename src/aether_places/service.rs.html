23/// Writes the places database from a dedicated thread, so that SQLite never blocks the caller.
24///
25/// Commands are applied in the order they are sent. Dropping the service waits until the
26/// pending ones are written.
27pub struct PlacesService {
28    /// Only `None` while the service is dropped, to close the channel before joining.
29    sender: Option<Sender<PlacesRequest>>,
30    cleaner: HistoryCleaner,
31    thread: Option<JoinHandle<()>>,
32    bookmarks: Receiver<SavedBookmarks>,
33}
34
35const HISTORY_UNAVAILABLE: &str = "The history is not available";
36
37type Reply = oneshot::Sender<Result<(), String>>;
38
39enum CleanerRequest {
40    SetRetention(Duration, Reply),
41    DeleteVisitsSince(SystemTime, Reply),
42}
43
44/// Sends the saved bookmarks to the owner of a [`PlacesService`], and tells it to take them.
45struct BookmarksPublisher {
46    sender: Sender<SavedBookmarks>,
47    notify: Box<dyn Fn() + Send>,
48}
49
50impl BookmarksPublisher {
51    fn publish(&self, urls: Vec<Url>, revision: u64) {
52        let _ = self.sender.send(SavedBookmarks { urls, revision });
53        (self.notify)();
54    }
55}
56
57/// The bookmarks stored in the database, and the newest bookmark change included in that read.
58///
59/// `revision` is zero when no bookmark change has been attempted yet.
60pub struct SavedBookmarks {
61    pub urls: Vec<Url>,
62    pub revision: u64,
63}
64
65struct PlacesRequest {
66    command: PlacesCommand,
67    /// The interface's revision of `command` when it changes bookmarks.
68    bookmark_revision: Option<u64>,
69}
70
71/// Deletes visits of a [`PlacesService`], after the changes already sent to it.
72#[derive(Clone)]
73pub struct HistoryCleaner(Sender<CleanerRequest>);
74
75impl HistoryCleaner {
76    /// Keeps visits for `retention`, and deletes the older ones now.
77    pub fn set_retention(&self, retention: Duration) -> CleanerReply {
78        self.request(|reply| CleanerRequest::SetRetention(retention, reply))
79    }
80
81    /// Deletes the visits at or after `date`.
82    pub fn delete_visits_since(&self, date: SystemTime) -> CleanerReply {
83        self.request(|reply| CleanerRequest::DeleteVisitsSince(date, reply))
84    }
85
86    fn request(&self, request: impl FnOnce(Reply) -> CleanerRequest) -> CleanerReply {
87        let (reply, receiver) = oneshot::channel();
88        let _ = self.0.send(request(reply));
89        CleanerReply(receiver)
90    }
91}
92
93/// Resolves when the deletion asked to a [`HistoryCleaner`] is done, with its error if any.
94#[must_use]
95pub struct CleanerReply(oneshot::Receiver<Result<(), String>>);
96
97impl Future for CleanerReply {
98    type Output = Result<(), String>;
99
100    fn poll(mut self: Pin<&mut Self>, context: &mut Context<'_>) -> Poll<Self::Output> {
101        Pin::new(&mut self.0)
102            .poll(context)
103            .map(|result| result.unwrap_or_else(|_| Err(HISTORY_UNAVAILABLE.to_owned())))
104    }
105}

107impl PlacesService {
108    /// Starts a service writing to the places database at `path`. `notify` is called from
109    /// another thread when [`Self::take_bookmarks`] has something new.
110    pub fn open(path: PathBuf, retention: Duration, notify: impl Fn() + Send + 'static) -> Self {
111        Self::start(move || PlacesStore::open(&path), retention, notify)
112    }
113
114    /// Starts a service whose database is never written to disk.
115    pub fn open_in_memory(retention: Duration, notify: impl Fn() + Send + 'static) -> Self {
116        Self::start(PlacesStore::open_in_memory, retention, notify)
117    }
118
119    fn start(
120        open_store: impl FnOnce() -> rusqlite::Result<PlacesStore> + Send + 'static,
121        retention: Duration,
122        notify: impl Fn() + Send + 'static,
123    ) -> Self {
124        let (sender, receiver) = unbounded();
125        let (cleaner_sender, cleaner_receiver) = unbounded();
126        let (bookmarks_sender, bookmarks) = unbounded();
127        let publisher = BookmarksPublisher {
128            sender: bookmarks_sender,
129            notify: Box::new(notify),
130        };
131        let thread = thread::Builder::new()
132            .name("Places".to_owned())
133            .spawn(move || match open_store() {
134                Ok(store) => {
135                    if let Some(urls) = read_bookmarks(&store) {
136                        publisher.publish(urls, 0);
137                    }
138                    run(store, receiver, retention, cleaner_receiver, publisher)
139                },
140                // Nothing is saved for this session, but browsing keeps working.
141                Err(error) => error!("Could not open the places database: {error}"),
142            })
143            .expect("Thread spawning failed");
144        Self {
145            sender: Some(sender),
146            cleaner: HistoryCleaner(cleaner_sender),
147            thread: Some(thread),
148            bookmarks,
149        }
150    }
151
152    /// Returns a handle to delete visits.
153    pub fn cleaner(&self) -> HistoryCleaner {
154        self.cleaner.clone()
155    }
156
157    /// Records a visit of `url` at `visit_date`.
158    pub fn record_visit(&self, url: Url, visit_date: SystemTime, title: Option<String>) {
159        self.send(PlacesCommand::RecordVisit {
160            url,
161            visit_date,
162            title,
163        });
164    }
165
166    /// Stores `title` as the last title of `url`.
167    pub fn set_title(&self, url: Url, title: String) {
168        self.send(PlacesCommand::SetTitle { url, title });
169    }
170
171    /// Moves the visit of `old_url` at `visit_date` to `new_url`, or deletes it when `new_url`
172    /// is `None`.
173    pub fn replace_visit(
174        &self,
175        old_url: Url,
176        visit_date: SystemTime,
177        new_url: Option<Url>,
178        title: Option<String>,
179    ) {
180        self.send(PlacesCommand::ReplaceVisit {
181            old_url,
182            visit_date,
183            new_url,
184            title,
185        });
186    }
187
188    /// The saved bookmarks, if they were sent since the last call: at startup and after a
189    /// failed write.
190    pub fn take_bookmarks(&self) -> Option<SavedBookmarks> {
191        self.bookmarks.try_iter().last()
192    }
193
194    /// Bookmarks `url`, unless it already is.
195    ///
196    /// `revision` identifies this change. A later read reports the newest revision it includes.
197    pub fn add_bookmark(
198        &self,
199        url: Url,
200        title: Option<String>,
201        date_added: SystemTime,
202        revision: u64,
203    ) {
204        self.send_request(PlacesRequest {
205            command: PlacesCommand::AddBookmark {
206                url,
207                title,
208                date_added,
209            },
210            bookmark_revision: Some(revision),
211        });
212    }
213
214    /// Removes the bookmark of `url`.
215    ///
216    /// `revision` identifies this change. A later read reports the newest revision it includes.
217    pub fn remove_bookmark(&self, url: Url, revision: u64) {
218        self.send_request(PlacesRequest {
219            command: PlacesCommand::RemoveBookmark { url },
220            bookmark_revision: Some(revision),
221        });
222    }
223
224    fn send(&self, command: PlacesCommand) {
225        self.send_request(PlacesRequest {
226            command,
227            bookmark_revision: None,
228        });
229    }
230
231    fn send_request(&self, request: PlacesRequest) {
232        if let Some(sender) = &self.sender {
233            let _ = sender.send(request);
234        }
235    }
236}

238impl Drop for PlacesService {
239    fn drop(&mut self) {
240        // Closing the channel ends the thread once it has applied the pending commands.
241        self.sender.take();
242        if let Some(thread) = self.thread.take() &&
243            thread.join().is_err()
244        {
245            error!("The places thread panicked");
246        }
247    }
248}

250fn read_bookmarks(store: &PlacesStore) -> Option<Vec<Url>> {
251    store
252        .bookmarked_urls()
253        .inspect_err(|error| warn!("Could not read the bookmarks: {error}"))
254        .ok()
255}
256
257fn run(
258    mut store: PlacesStore,
259    receiver: Receiver<PlacesRequest>,
260    mut retention: Duration,
261    cleaner_receiver: Receiver<CleanerRequest>,
262    publisher: BookmarksPublisher,
263) {
264    let expiration_timer = tick(EXPIRATION_INTERVAL);
265    expire_or_warn(&mut store, retention);
266    loop {
267        select! {
268            recv(receiver) -> request => {
269                let Ok(request) = request else {
270                    for cleaner_request in cleaner_receiver.try_iter() {
271                        clean(&mut store, cleaner_request, &mut retention);
272                    }
273                    return;
274                };
275                let requests = iter::once(request).chain(receiver.try_iter()).collect();
276                write(&mut store, requests, &publisher);
277            },
278            recv(cleaner_receiver) -> request => {
279                let Ok(request) = request else {
280                    return;
281                };
282                write(&mut store, receiver.try_iter().collect(), &publisher);
283                clean(&mut store, request, &mut retention);
284            },
285            recv(expiration_timer) -> _ => expire_or_warn(&mut store, retention),
286        }
287    }
288}
289
290fn clean(store: &mut PlacesStore, request: CleanerRequest, retention: &mut Duration) {
291    let (result, reply) = match request {
292        CleanerRequest::SetRetention(new_retention, reply) => {
293            *retention = new_retention;
294            (expire(store, *retention), reply)
295        },
296        CleanerRequest::DeleteVisitsSince(date, reply) => (store.delete_visits_since(date), reply),
297    };
298    let result = result.map_err(|error| {
299        warn!("Could not delete the history: {error}");
300        format!("Could not delete the history: {error}")
301    });
302    let _ = reply.send(result);
303}
304
305fn write(store: &mut PlacesStore, requests: Vec<PlacesRequest>, publisher: &BookmarksPublisher) {
306    if requests.is_empty() {
307        return;
308    }
309    let revision = requests
310        .iter()
311        .filter_map(|request| request.bookmark_revision)
312        .max();
313    let commands: Vec<_> = requests
314        .into_iter()
315        .map(|request| request.command)
316        .collect();
317    if let Err(error) = store.apply(&commands) {
318        warn!(
319            "Could not write {} changes to the places database: {error}",
320            commands.len()
321        );
322        // The revision is the newest bookmark change of this attempt. The read is the database
323        // without that attempt, so the interface can keep the changes it sent afterwards.
324        if let Some(revision) = revision &&
325            let Some(urls) = read_bookmarks(store)
326        {
327            publisher.publish(urls, revision);
328        }
329    }
330}
331
332fn expire(store: &mut PlacesStore, retention: Duration) -> rusqlite::Result<()> {
333    match SystemTime::now().checked_sub(retention) {
334        Some(date) => store.delete_visits_before(date),
335        None => Ok(()),
336    }
337}
338
339fn expire_or_warn(store: &mut PlacesStore, retention: Duration) {
340    if let Err(error) = expire(store, retention) {
341        warn!("Could not delete the expired history: {error}");
342    }
343}
